primary_c2_receiver.py 
1    #!/usr/bin/env python3
2    import socket
3    import ast
4    import threading
5    
6    # Direct telemetry configuration
7    HOST = "0.0.0.0"  # Listen on all network interfaces
8    DIRECT_PORT = 9999  # Port for direct telemetry (including relayed data)
9    
10   # Multicast configuration for UAV-to-UAV communication
11   MULTICAST_GROUP = "239.255.255.250"
12   MULTICAST_PORT = 5000
13   
14   
15   def receive_direct_telemetry():
16       """Receive direct (and relayed) telemetry from UAVs on port 9999."""
17       sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
18       sock.bind((HOST, DIRECT_PORT))
19       print(f"Command Center listening for direct telemetry on port {DIRECT_PORT}")
20       while True:
21           data, addr = sock.recvfrom(1024)
22           try:
23               telemetry = ast.literal_eval(data.decode('utf-8'))
24           except Exception as e:
25               print("Error parsing direct telemetry:", e)
26               telemetry = {"raw_data": data.decode('utf-8')}
27           print("\n--- Direct Telemetry ---")
28           print(f"From {addr}:")
29           print(f"  UAV ID  : {telemetry.get('id', 'Unknown')}")
30           print(f"  Lat/Lon : {telemetry.get('lat', 'N/A')}, {telemetry.get('lon', 'N/A')}")
31           print(f"  Alt     : {telemetry.get('alt', 'N/A')} m")
32           print(f"  Battery : {telemetry.get('battery', 'N/A')}%")
33           print(f"  Channel : {telemetry.get('channel', 'None')}")
34   
35   
36   def receive_multicast_telemetry():
37       """Listen for multicast telemetry directly from UAVs."""
38       sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM, socket.IPPROTO_UDP)
39       sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
40       sock.bind(('', MULTICAST_PORT))
41   
42       mreq = socket.inet_aton(MULTICAST_GROUP) + socket.inet_aton("0.0.0.0")
43       sock.setsockopt(socket.IPPROTO_IP, socket.IP_ADD_MEMBERSHIP, mreq)
44       print(f"Command Center listening for multicast telemetry on {MULTICAST_GROUP}:{MULTICAST_PORT}")
45   
46       while True:
47           data, addr = sock.recvfrom(1024)
48           try:
49               telemetry = ast.literal_eval(data.decode('utf-8'))
50           except Exception as e:
51               print("Error parsing multicast telemetry:", e)
52               telemetry = {"raw_data": data.decode('utf-8')}
53           print("\n+++ Multicast Telemetry +++")
54           print(f"From {addr}:")
55           print(f"  UAV ID  : {telemetry.get('id', 'Unknown')}")
56           print(f"  Lat/Lon : {telemetry.get('lat', 'N/A')}, {telemetry.get('lon', 'N/A')}")
57           print(f"  Alt     : {telemetry.get('alt', 'N/A')} m")
58           print(f"  Battery : {telemetry.get('battery', 'N/A')}%")
59           print(f"  Channel : {telemetry.get('channel', 'None')}")
60   
61   
62   if __name__ == "__main__":
63       # Start threads to concurrently listen for both direct and multicast telemetry.
64       direct_thread = threading.Thread(target=receive_direct_telemetry, daemon=True)
65       multicast_thread = threading.Thread(target=receive_multicast_telemetry, daemon=True)
66   
67       direct_thread.start()
68       multicast_thread.start()
69   
70       # Keep the main thread running.
71       try:
72           while True:
73               pass
74       except KeyboardInterrupt:
75           print("\nCommand Center shutting down.")
