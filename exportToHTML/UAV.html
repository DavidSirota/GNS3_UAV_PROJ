1    #!/usr/bin/env python3
2    import socket
3    import time
4    import random
5    
6    # Configuration
7    C2_IP = "10.0.2.10"    # IP of the primary command control center
8    C2_PORT = 9999         # Port for receiving telemetry
9    UAV_ID = "UAV1"        # Change for each UAV instance
10   
11   def generate_telemetry():
12       # Simulate some telemetry: latitude, longitude, altitude, battery level
13       telemetry = {
14           "id": UAV_ID,
15           "lat": round(40.7128 + random.uniform(-0.05, 0.05), 4),
16           "lon": round(-74.0060 + random.uniform(-0.05, 0.05), 4),
17           "alt": random.randint(100, 500),  # altitude in meters
18           "battery": random.randint(20, 100)  # battery percentage
19       }
20       return str(telemetry)
21   
22   def send_telemetry():
23       with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as sock:
24           while True:
25               message = generate_telemetry()
26               sock.sendto(message.encode('utf-8'), (C2_IP, C2_PORT))
27               print(f"{UAV_ID} sent: {message}")
28               time.sleep(5)  # send telemetry every 5 seconds
29   
30   if __name__ == "__main__":
31       try:
32           send_telemetry()
33       except KeyboardInterrupt:
34           print("Telemetry simulation stopped.")
